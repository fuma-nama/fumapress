### Replacing a Default Plugin

A site replaces a [default plugin](/docs/plugins#replacing-a-default-plugin) by setting it to `false` and listing yours. This image optimizer wraps every page in its provider:

```tsx tab="press.config.tsx"
import { createPress, recommendedPreset, type WakuAppShape, type WakuPlugin } from "fumapress";
import { CloudinaryProvider } from "./src/provider";

function cloudinaryImage<C extends WakuAppShape = WakuAppShape>(options: {
  cloud: string;
}): WakuPlugin<C, { slots: "cloudinary:provider" }> {
  return {
    name: "cloudinary:image",
    slots: ({ page }) =>
      page
        ? undefined
        : [
            {
              name: "cloudinary:provider",
              region: "layout-wrapper",
              component: CloudinaryProvider,
              props: { cloud: options.cloud },
              order: -10,
            },
          ],
  };
}

export const app = createPress(content).plugins(cloudinaryImage({ cloud: "acme" }));

export default app.config({
  preset: recommendedPreset({ image: false }), // [!code highlight]
});
```

```tsx tab="src/provider.tsx"
"use client";
import type { ReactNode } from "react";

export function CloudinaryProvider({ cloud, children }: { cloud: string; children?: ReactNode }) {
  return <div data-cloud={cloud}>{children}</div>;
}
```
