### Replacing a Default Plugin

A plugin replaces a [default plugin](/docs/plugins#default-plugins) when both have the same `provides` value. This image optimizer provides `image`, and its provider wraps every layout:

```tsx tab="press.config.tsx"
import { createPress, type WakuAppShape, type WakuPlugin } from "fumapress";
import { CloudinaryProvider } from "./src/provider";

function cloudinaryImage<C extends WakuAppShape = WakuAppShape>(options: {
  cloud: string;
}): WakuPlugin<C, { provides: "image" }> {
  return {
    name: "cloudinary:image",
    provides: ["image"], // [!code highlight]
    slots: ({ page }) =>
      page
        ? undefined
        : [
            {
              slot: "layout.wrapper",
              component: CloudinaryProvider,
              props: { cloud: options.cloud },
              order: -10,
            },
          ],
  };
}

export default createPress(content).plugins(cloudinaryImage({ cloud: "acme" }));
```

```tsx tab="src/provider.tsx"
"use client";
import type { ReactNode } from "react";

export function CloudinaryProvider({ cloud, children }: { cloud: string; children?: ReactNode }) {
  return <div data-cloud={cloud}>{children}</div>;
}
```
