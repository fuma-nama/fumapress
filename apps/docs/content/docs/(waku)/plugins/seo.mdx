```tsx title="press.config.tsx"
  takumiPlugin({ width: 1600, height: 840 }),
);
```

### Custom Pages

Custom pages and plugin pages are listed with priority `1`. To leave a page out or change its entry, set `sitemap` in `getConfig()`:

```tsx title="src/pages/preview.tsx"
export function getConfig() {
  return { sitemap: false };
}
```

A plugin passes the same option to `createPage()` in `waku.createPages`.

### Custom Pages

To give a custom page an image, set `ogImage` in `getConfig()`. It uses the default design unless you pass `node`. For a page with params, use a function:

```tsx tab="src/pages/pricing.tsx"
import type { WakuRouteConfig } from "fumapress";

export function getConfig() {
  return { render: "static", ogImage: { title: "Pricing" } } satisfies WakuRouteConfig;
}
```

```tsx tab="src/pages/showcase/[slug].tsx"
import type { WakuRouteConfig } from "fumapress";

export function getConfig() {
  return {
    render: "static",
    staticPaths: Object.keys(items),
    ogImage(params) {
      const item = items[params.slug as string];
      return { title: item?.title, description: item?.description };
    },
  } satisfies WakuRouteConfig;
}
```

### WebAssembly Build

When the machine building your site can't load the native addon of Takumi, pass the WebAssembly build. On Cloudflare Workers, it is used automatically.

```tsx title="press.config.tsx"
import { takumiPlugin } from "fumapress/plugins/takumi";
import wasm from "fumapress/plugins/takumi.wasm"; // [!code ++]

takumiPlugin({
  options: {
    module: wasm, // [!code ++]
  },
});
```
