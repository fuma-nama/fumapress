## Custom Layout

The default layout places the form for you. With a custom [`renderPage`](/docs/layouts#page-actions-and-footers), place `footer.feedback`:

```tsx title="src/ui/layout.tsx"
function Docs(props: RenderPageProps<typeof app, "docs">) {
  return (
    <DocsLayout {...props.baseOptions} tree={props.tree}>
      <DocsPage toc={props.pageData.toc}>
        <DocsTitle>{props.pageData.title}</DocsTitle>
        <DocsBody>{props.body}</DocsBody>
        {props.footer.feedback} {/* [!code highlight] */}
      </DocsPage>
    </DocsLayout>
  );
}
```
