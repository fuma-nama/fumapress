## Overview

The recommended preset includes [FlexSearch](/docs/plugins/flexsearch) by default. In static mode, the dialog downloads the search index from `/api/search` and searches in the browser. Otherwise it queries `/api/search` on each keystroke.

One search plugin per site: it provides the `search` capability.

## Search Engines

List `oramaSearchPlugin()` to replace the preset's FlexSearch with [Orama](/docs/plugins/orama-search):

```tsx title="press.config.tsx"
import { createPress } from "fumapress";
import { oramaSearchPlugin } from "fumapress/plugins/orama-search";
// ...

export default createPress(docs.toFumadocsSource())
  .plugins(oramaSearchPlugin())
  .config({ site: { name: "Acme" } });
```

To disable search, filter the preset entry:

```tsx title="press.config.tsx"
import { createPress, recommendedPreset } from "fumapress";
// ...

export default createPress(docs.toFumadocsSource()).config({
  preset: recommendedPreset().filter((entry) => entry.provides !== "search"),
});
```

## Custom Dialog

`ui.providerProps.search.SearchDialog` overrides the search dialog. It receives the plugin's `search.options` as props, like `from` and `locale` for the static FlexSearch index. Compose it with the dialog parts of `fumadocs-ui/components/dialog/search` and a [Fumadocs search hook](https://fumadocs.dev/docs/headless/search).

```tsx tab="src/search.tsx"
"use client";
import {
  useFlexsearchStatic,
  type FlexsearchStaticOptions,
} from "fumadocs-core/search/client/flexsearch-static";
import {
  SearchDialog as Dialog,
  SearchDialogContent,
  SearchDialogHeader,
  SearchDialogInput,
  SearchDialogList,
  SearchDialogOverlay,
  type SharedProps,
} from "fumadocs-ui/components/dialog/search";

export function SearchDialog({ from, locale, ...props }: SharedProps & FlexsearchStaticOptions) {
  return (
    <Dialog {...props} {...useFlexsearchStatic({ from, locale, tag: "guides" })}>
      <SearchDialogOverlay />
      <SearchDialogContent>
        <SearchDialogHeader>
          <SearchDialogInput />
        </SearchDialogHeader>
        <SearchDialogList />
      </SearchDialogContent>
    </Dialog>
  );
}
```

```tsx tab="press.config.tsx"
import { createPress } from "fumapress";
import { defineDocs } from "fumadocs-mdx/macro";
import { SearchDialog } from "./src/search";

const docs = defineDocs({ dir: "content/docs" });

export default createPress({ docs: docs.toFumadocsSource() }).config({
  mode: "static",
  ui: { providerProps: { search: { SearchDialog } } },
});
```

## Search Providers

A search provider is a plugin with `provides: ["search"]`. It returns its dialog and options from `providerProps({ lang })`, and adds its own API routes. Listing it takes the place of the preset's FlexSearch.

```tsx tab="acme/index.ts"
import type { AppShape, Plugin } from "@fumapress/core";
import { AcmeSearchDialog } from "./client";

export function acmeSearchPlugin<C extends AppShape = AppShape>(options: {
  index: string;
}): Plugin<C, { provides: "search" }> {
  return {
    name: "acme:search",
    provides: ["search"],
    providerProps({ lang }) {
      const api = this.withBase("/api/acme-search");
      return { search: { SearchDialog: AcmeSearchDialog, options: { api, locale: lang } } };
    },
    apiRoutes() {
      return [
        {
          path: "/api/acme-search",
          render: "dynamic",
          GET: (request) =>
            fetch(`https://search.acme.dev/${options.index}${new URL(request.url).search}`),
        },
      ];
    },
  };
}
```

```tsx tab="acme/client.tsx"
"use client";
import { useFetchSearch, type FetchOptions } from "fumadocs-core/search/client/fetch";
import {
  SearchDialog,
  SearchDialogContent,
  SearchDialogHeader,
  SearchDialogInput,
  SearchDialogList,
  SearchDialogOverlay,
  type SharedProps,
} from "fumadocs-ui/components/dialog/search";

export function AcmeSearchDialog({ api, locale, ...props }: SharedProps & FetchOptions) {
  return (
    <SearchDialog {...props} {...useFetchSearch({ api, locale })}>
      <SearchDialogOverlay />
      <SearchDialogContent>
        <SearchDialogHeader>
          <SearchDialogInput />
        </SearchDialogHeader>
        <SearchDialogList />
      </SearchDialogContent>
    </SearchDialog>
  );
}
```

```tsx tab="press.config.tsx"
import { createPress } from "fumapress";
import { defineDocs } from "fumadocs-mdx/macro";
import { acmeSearchPlugin } from "./acme";

const docs = defineDocs({ dir: "content/docs" });

export default createPress({ docs: docs.toFumadocsSource() })
  .plugins(acmeSearchPlugin({ index: "docs" }))
  .config({ mode: "dynamic" });
```

See [Capabilities](/docs/plugins/authoring#capabilities) for declaring what a plugin provides.
