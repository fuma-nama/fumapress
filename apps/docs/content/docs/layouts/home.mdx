## Plugin Views

Plugin views and plugin content pages (blog, changelog) render in the Home Layout by default. To restyle them, add a `renderPage` case with `HomeLayout` around `props.children`:

```tsx title="src/ui/layout.tsx"
import type { ReactNode } from "react";
import type { BaseOptions, RenderPageProps } from "fumapress";
import { HomeLayout } from "fumadocs-ui/layouts/home";
import type { app } from "../../press.config";

function HomeShell({ baseOptions, children }: { baseOptions: BaseOptions; children: ReactNode }) {
  return (
    <HomeLayout {...baseOptions}>
      <main className="mx-auto w-full max-w-[1400px] px-4 py-6">{children}</main>
    </HomeLayout>
  );
}

export function renderPage(props: RenderPageProps<typeof app>) {
  const { renderDefault } = props;
  switch (props.type) {
    case "blog":
    case "blog.index":
    case "blog.tags":
    case "blog.tag":
      return <HomeShell baseOptions={props.baseOptions}>{props.children}</HomeShell>;
    default:
      return renderDefault();
  }
}
```

See [Blog Custom UI](/docs/plugins/blog#custom-ui) for replacing the blog's views with your own components.

## Your Own Pages

Waku only. Pages in `src/pages` use `_layout.tsx` to wrap children with the Home Layout. The site context provides base options matching your config:

```tsx title="src/pages/_layout.tsx"
import type { ReactNode } from "react";
import { HomeLayout } from "fumadocs-ui/layouts/home";
import { getPressContext, type WakuRouteProps } from "fumapress";
import type { app } from "../../press.config";

export default async function Layout({
  lang,
  children,
}: WakuRouteProps<typeof app> & { children: ReactNode }) {
  const ctx = getPressContext<typeof app>();
  return <HomeLayout {...await ctx.getBaseOptions({ lang })}>{children}</HomeLayout>;
}
```

See [File-based Routing](/docs/routing#file-based-routing) for the route conventions.
