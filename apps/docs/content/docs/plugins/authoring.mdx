## Overview

A plugin is a generic factory returning `Plugin<C, T>`. Import it from `fumapress` in a site, or from `@fumapress/core` in a package that supports both runtimes. `this` in hooks is the site context.

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "@fumapress/core";

type StatusTypes = {
  provides: "acme:status";
  views: { "status.page": { ok: boolean } };
};

export function statusPlugin<C extends AppShape = AppShape>(): Plugin<C, StatusTypes> {
  return {
    name: "acme:status",
    provides: ["acme:status"],
    routes({ createApi, createView }) {
      createView({
        path: "/status",
        render: "dynamic",
        view: () => ({
          type: "status.page",
          props: { ok: true },
          node: <p>OK</p>,
        }),
      });
      createApi({ path: "/status.json", GET: () => Response.json({ ok: true }) });
    },
    head({ page, route }) {
      if (page || route) return;
      return [{ tag: "meta", attrs: { name: "generator", content: "acme" } }];
    },
    async middleware(request, next) {
      const response = await next();
      const out = new Response(response.body, response);
      out.headers.set("x-powered-by", "acme");
      return out;
    },
  };
}
```

The same value works on Waku and Astro:

```tsx title="press.config.tsx"
import { createPress } from "fumapress";

export default createPress({ docs: docs.toFumadocsSource() })
  .plugins(statusPlugin())
  .config({
    mode: "dynamic",
    ui: {
      renderPage(props) {
        if (props.type === "status.page") return <main>{props.ok ? "up" : "down"}</main>;
        return props.renderDefault();
      },
    },
  });
```

## Views

`createView()` in the `routes` hook registers pages rendered in the [Home Layout](https://fumadocs.dev/docs/ui/layouts/home). A view is created in every language and receives its `lang`. Set `lang` to create it in one language, like a path that differs per language. The site prefixes `path` with the language segment and links the pages of other languages as translations. `ui.renderPage` renders each view as `props.type`.

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "@fumapress/core";

export interface ShowcaseItem {
  name: string;
  url: string;
}

export type ShowcaseTypes = {
  views: { "showcase.index": { items: ShowcaseItem[] } };
  actions: "showcase";
};

export function showcasePlugin<C extends AppShape = AppShape>(options: {
  items: ShowcaseItem[];
  path?: string;
}): Plugin<C, ShowcaseTypes> {
  const path = options.path ?? "/showcase";
  return {
    name: "acme:showcase",
    routes({ createView }) {
      createView({
        path,
        view: () => ({
          type: "showcase.index",
          props: { items: options.items },
          node: (
            <ul>
              {options.items.map((item) => (
                <li key={item.url}>{item.name}</li>
              ))}
            </ul>
          ),
          title: "Showcase",
        }),
      });
    },
    // ...
  };
}
```

The site renders the view with `ui.renderPage`:

```tsx title="press.config.tsx"
import { createPress } from "fumapress";

export default createPress({ docs: docs.toFumadocsSource() })
  .plugins(showcasePlugin({ items: [{ name: "Fumapress", url: "https://press.fumadocs.dev" }] }))
  .config({
    ui: {
      renderPage(props) {
        const { renderDefault } = props;
        switch (props.type) {
          case "showcase.index":
            return (
              <section>
                {props.items.length} sites
                {props.children}
              </section>
            );
          default:
            return renderDefault();
        }
      },
    },
  });
```

A view rendered on request (`render: "dynamic"`) receives the request. Return `undefined` for a 404:

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "@fumapress/core";

type AccountTypes = { views: { "acme.account": { user: string } } };

export function accountPlugin<C extends AppShape = AppShape>(): Plugin<C, AccountTypes> {
  return {
    name: "acme:account",
    routes({ createView }) {
      createView({
        path: "/account",
        render: "dynamic",
        view(_params, { request }) {
          const user = /session=([^;]+)/.exec(request.headers.get("cookie") ?? "")?.[1];
          if (!user) return;
          return {
            type: "acme.account",
            props: { user },
            node: <p>Signed in as {user}</p>,
            title: "Account",
          };
        },
      });
    },
  };
}
```

## API Routes

`createApi()` in the `routes` hook creates a route with method handlers. It is handled on the server. Set `render: "static"` to generate a file from its `GET` handler at build time, listing its params in `staticPaths`:

```tsx title="acme/index.tsx"
routes({ createApi }) {
  createApi({ path: "/status.json", GET: () => Response.json({ ok: true }) });
  createApi({
    path: "/data/[id].json",
    render: "static",
    staticPaths: ["a", "b"],
    GET: (_request, { params }) => Response.json({ id: params.id }),
  });
},
```

A static site can't serve routes handled on the server, and the build fails naming the route. Read `this.mode` to degrade.

## Derived Routes

`deriveRoutes(fns, { lang, routes })` creates a route per page from the final page routes of a language. Use it for print pages or per-page Markdown:

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "@fumapress/core";

type PrintTypes<P> = { views: { print: { page: P } } };

export function printPlugin<C extends AppShape = AppShape>(): Plugin<C, PrintTypes<C["page"]>> {
  return {
    name: "acme:print",
    async deriveRoutes({ createView }, { lang }) {
      const loader = await this.getLoader();
      createView({
        path: "/print/[...slugs]",
        staticPaths: loader.getPages(lang).map((page) => page.slugs),
        view: async ({ slugs }) => {
          const page = loader.getPage(slugs as string[], lang);
          if (!page) return;
          const body = await this.getPageBody(page);
          return {
            type: "print",
            props: { page },
            node: body?.node,
            title: page.data.title,
          };
        },
      });
    },
  };
}
```

The path has no language prefix; the site adds it. A view without `lang` is created in the language of the call.

## Page Views

`getPageView(page, body)` renders content pages of a source the plugin owns. `body` is the page content with `body.wrapper` applied. Declare the props under `pages` in the type parameter.

See [Blog](/docs/plugins/blog) for a complete example. For the rendering hooks (`getPageBody`, `getPageText`, etc.), see [Content Hooks](/docs/adapters#content-hooks).

## Slots

`slots({ page })` adds UI to layouts. Named slots (`actions` and `footer`) appear in `ui.renderPage` as `props.actions[name]` and `props.footer[name]`. Wrapper slots (`body.wrapper` and `layout.wrapper`) receive their content as `children`. `order` controls nesting: lower is outer.

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "@fumapress/core";
import { ShowcaseAction } from "./action";

export type ShowcaseTypes = {
  views: { "showcase.index": { items: ShowcaseItem[] } };
  actions: "showcase";
};

export function showcasePlugin<C extends AppShape = AppShape>(options: {
  items: ShowcaseItem[];
  path?: string;
}): Plugin<C, ShowcaseTypes> {
  const path = options.path ?? "/showcase";
  return {
    name: "acme:showcase",
    // ...
    slots({ page }) {
      if (!page) return;
      return [
        {
          slot: "actions",
          name: "showcase",
          component: ShowcaseAction,
          props: { href: this.withBase(path) },
        },
      ];
    },
  };
}
```

The `props` of a client component must be serializable.

A `layout.wrapper` item wraps every page, like the site-wide provider of [Inline Plugins](/docs/plugins#inline-plugins).

## Base Options and Provider Props

`baseOptions({ lang })` contributes [base options](/docs/layouts#base-options) merged under the site's. Links concatenate and objects merge deeply. `providerProps({ lang })` contributes provider props the same way.

```tsx title="acme/index.tsx"
baseOptions({ lang }) {
  return {
    links: [{ url: this.localizePath(lang, path), text: "Showcase" }],
  };
},
```

## Head Tags

`head({ page, route, path, view })` adds tags to `<head>`. The site-wide call has no `page`, `route` or `path`. Per-page calls include them. The site's tags win over plugins' by key.

```tsx title="acme/index.tsx"
head({ page, route }) {
  if (page || route) return;
  return [{ tag: "meta", attrs: { name: "generator", content: "acme" } }];
},
```

## Middleware

`middleware(request, next)` runs before routes on the server. Call `next()` to continue, or `next(path)` to serve another path instead. Middleware requires a server; it does not apply to static deployments.

```tsx title="acme/index.tsx"
import type { AppShape, Plugin } from "fumapress";

function guardPlugin<C extends AppShape = AppShape>(options: {
  prefix: string;
  token: string;
}): Plugin<C> {
  return {
    name: "acme:guard",
    async middleware(request, next) {
      const { pathname } = new URL(request.url);
      if (
        pathname.startsWith(this.withBase(options.prefix)) &&
        request.headers.get("authorization") !== `Bearer ${options.token}`
      )
        return new Response("Unauthorized", { status: 401 });
      return next();
    },
  };
}
```

## Capabilities

`provides` declares the capabilities a plugin provides. `api` exposes a typed API that other plugins read with `this.getCapability(name)`. Augment the `CapabilityAPI` interface of `@fumapress/core` to type your capability.

A search provider declares `provides: ["search"]` and puts its dialog in `providerProps`. See [Search Providers](/docs/search#search-providers).

## Lifecycle

`configure()` runs after the content loader is created. Use `this.signal` to stop work when the runtime drops the context, and `this.invalidateLoader(source)` to reload content:

```ts title="acme/notes-plugin.ts"
import { watch } from "node:fs";
import type { AppShape, Plugin } from "fumapress";

export function notesPlugin<C extends AppShape = AppShape>(options: {
  dir: string;
  source: C["source"];
}): Plugin<C> {
  return {
    name: "acme:notes",
    configure() {
      if (!import.meta.env.DEV) return;
      watch(options.dir, { recursive: true, signal: this.signal }, () =>
        this.invalidateLoader(options.source),
      );
    },
  };
}
```

`build(output)` runs once after the site is built. Throw to fail the build.

`transformPage` returns the changed page data fields. A field has one writer across all plugins.

## Types

`Plugin<C, T>` takes a second type parameter `T` declaring what the plugin adds to the site. It types `props.type`, the slot items and `this.getCapability(...)`:

```ts title="acme/index.tsx"
type AcmeTypes = {
  // view types and their props in `ui.renderPage`
  views: { "showcase.index": { items: ShowcaseItem[] } };
  // props added to the content pages of its sources
  pages: { rating: number };
  // its capabilities
  provides: "acme:status";
  // `props.actions.showcase` and `props.footer.rating`
  actions: "showcase";
  footer: "rating";
};
```

## Waku Hooks

Waku only. `WakuPlugin<C>` adds hooks under `waku` for the Waku router and server. `C` extends `WakuAppShape`, so Astro sites refuse the plugin by type.

`waku.createPages` creates pages and layouts with `createPage`, `createLayout` and `createSlice`. Each is registered once per language with `lang`:

```tsx title="acme/index.tsx"
import type { ReactNode } from "react";
import { Slice } from "waku/router/client";
import type { WakuAppShape, WakuPlugin } from "fumapress";

export interface ShowcaseOptions {
  items: Record<string, { title: string }>;
  analyticsId: string;
}

function ShowcaseLayout({ children }: { children: ReactNode }) {
  return <section data-showcase="">{children}</section>;
}

async function Stars() {
  const response = await fetch("https://api.github.com/repos/fuma-nama/fumadocs");
  return <span>{response.status}</span>;
}

export function showcasePlugin<C extends WakuAppShape = WakuAppShape>(
  options: ShowcaseOptions,
): WakuPlugin<C> {
  return {
    name: "acme:showcase",
    routes: ({ createApi }) => createApi({ path: "/acme/ping", GET: () => new Response("pong") }),
    waku: {
      createPages({ createSlice, createLayout, createPage }) {
        createSlice({ id: "acme-stars", render: "dynamic", component: Stars });
        for (const lang of this.languages) {
          createLayout({
            path: "/(showcase)",
            lang,
            render: "static",
            component: ShowcaseLayout,
          });
          createPage({
            path: "/(showcase)/showcase/[item]",
            lang,
            render: "static",
            staticPaths: Object.keys(options.items),
            slices: ["acme-stars"],
            component: ({ item }: { item: string }) => (
              <main>
                <h1>{options.items[item]?.title}</h1>
                <Slice id="acme-stars" lazy fallback={null} />
              </main>
            ),
          });
        }
      },
      honoMiddleware() {
        return async (c, next) => {
          await next();
          const kv: { put(key: string, value: string): Promise<void> } | undefined = c.env?.ACME_KV;
          await kv?.put(`hit:${c.req.path}`, options.analyticsId);
        };
      },
      unstable_serverEntry(entry) {
        return {
          ...entry,
          buildOptions: {
            ...entry.buildOptions,
            ACME_ANALYTICS_ID: options.analyticsId,
          },
        };
      },
    },
  };
}
```

Inline Waku hooks in the site config:

```tsx title="press.config.tsx"
import { createPress } from "fumapress";

export const app = createPress({ docs: docs.toFumadocsSource() }, { i18n }).plugins(
  showcasePlugin({ items: { fumadocs: { title: "Fumadocs" } }, analyticsId: "a1" }),
  {
    name: "site:waku",
    waku: {
      createPages({ createPage }) {
        createPage({
          path: "/status",
          lang: "zh",
          render: "dynamic",
          component: () => <main>OK</main>,
        });
      },
    },
  },
);

export default app.config({ mode: "dynamic" });
```
