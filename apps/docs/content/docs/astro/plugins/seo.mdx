```tsx title="press.config.tsx"
import { sitemapPlugin } from "@fumapress/astro/plugins/sitemap";
import { robotsPlugin } from "@fumapress/astro/plugins/robots";
import { rssPlugin } from "@fumapress/astro/plugins/rss";
import { takumiPlugin } from "@fumapress/astro/plugins/takumi";

createPress(content).plugins(
  sitemapPlugin({ path: "/sitemap.xml" }),
  robotsPlugin({ sitemap: false }),
  rssPlugin({ title: "My Docs", limit: 50 }),
  takumiPlugin({ width: 1600, height: 840 }),
);
```
