### Add the Collection [step]

Add a [collection](/docs/astro/content-sources/collections#add-a-collection) for them, with the changelog schema, and pass its key to the site plugin:

```js tab="astro.config.mjs"
press({
  collections: {
    docs: { dir: "content/docs" },
    changelog: { dir: "content/changelog" }, // [!code ++]
  },
});
```

```ts tab="src/content.config.ts"
import { defineCollection } from "astro:content";
import { metaLoader, pageLoader } from "@fumadocs/astro/loaders";
// [!code ++]
import { changelogPageSchema } from "@fumapress/tegami/schema";

export const collections = {
  docs: defineCollection({ loader: pageLoader() }),
  meta: defineCollection({ loader: metaLoader() }),
  // [!code ++:2]
  changelog: defineCollection({ loader: pageLoader("changelog"), schema: changelogPageSchema }),
  changelogMeta: defineCollection({ loader: metaLoader("changelog") }),
};
```

```tsx tab="press.config.tsx"
import { getCollection } from "astro:content";
import { toFumadocsSource } from "@fumadocs/astro/source";
import { createPress } from "@fumapress/astro/config";
// [!code ++]
import { changelogPlugin } from "@fumapress/tegami";

export default createPress({
  docs: toFumadocsSource(await getCollection("docs"), await getCollection("meta"), {
    baseDir: "docs",
  }),
  // [!code ++:5]
  changelog: toFumadocsSource(
    await getCollection("changelog"),
    await getCollection("changelogMeta"),
    { baseDir: "changelog" },
  ),
})
  // [!code ++]
  .plugins(changelogPlugin({ source: "changelog" }));
```
