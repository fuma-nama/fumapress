This guide creates by hand the files that `npm create fumapress` generates.

<show when={props.framework === "astro"}>

<Callout>Astro needs Node.js 22.23.4 or later.</Callout>

</show>

## Setup

### Install Dependencies [step]

<show when={props.framework === "waku"}>

```npm
npm i fumapress
npm i fumadocs-core fumadocs-mdx
npm i fumadocs-ui@npm:@fumadocs/base-ui @base-ui/react
npm i react react-dom

npm i vite tailwindcss @tailwindcss/vite -D
npm i typescript @types/mdx @types/node @types/react @types/react-dom -D
```

</show>

<show when={props.framework === "astro"}>

```npm
npm i @fumapress/astro
npm i @fumadocs/astro astro
npm i @astrojs/react react react-dom takumi-js fumadocs-core
npm i fumadocs-ui@npm:@fumadocs/base-ui

npm i tailwindcss @tailwindcss/vite -D
npm i typescript @types/mdx @types/react @types/react-dom -D
```

</show>

Fumapress uses the [Base UI build](https://fumadocs.dev/docs/ui/component-library) of Fumadocs UI, installed as `fumadocs-ui`.

### Create the Config Files [step]

<show when={props.framework === "waku"}>

```ts tab="vite.config.ts"
import { defineConfig } from "vite";
import tailwindcss from "@tailwindcss/vite";
import press from "fumapress/vite";
import { fumadocsMdx } from "fumadocs-mdx/vite";

export default defineConfig({
  plugins: [press(), fumadocsMdx(), tailwindcss()],
});
```

```tsx tab="press.config.tsx"
import { createPress } from "fumapress";
import { metaSchema, pageSchema } from "fumapress/schema";
import { defineDocs } from "fumadocs-mdx/macro";

const docs = defineDocs({
  dir: "content",
  docs: {
    async: true,
    schema: pageSchema,
    lastModified: true,
    postprocess: {
      includeProcessedMarkdown: true,
    },
  },
  meta: {
    schema: metaSchema,
  },
});

export default createPress(docs.toFumadocsSource()).config({
  site: {
    name: "My Docs",
    // the URL of your deployed site, for the sitemap and other full URLs
    // baseUrl: "https://example.com",
  },
});
```

```css tab="src/app.css"
@import "tailwindcss";
@import "fumadocs-ui/css/neutral.css";
@import "fumadocs-ui/css/preset.css";
@import "fumapress/css/preset.css";
```

```json tab="tsconfig.json"
{
  "compilerOptions": {
    "target": "ES2023",
    "lib": ["dom", "dom.iterable", "ES2023"],
    "jsx": "react-jsx",
    "module": "ESNext",
    "moduleResolution": "bundler",
    "resolveJsonModule": true,
    "allowJs": true,
    "checkJs": true,
    "esModuleInterop": true,
    "forceConsistentCasingInFileNames": true,
    "strict": true,
    "skipLibCheck": true,
    "noUncheckedIndexedAccess": true,
    "noEmit": true,
    "paths": { "@/*": ["./src/*"] }
  },
  "exclude": ["node_modules", "dist"]
}
```

</show>

<show when={props.framework === "astro"}>

```js tab="astro.config.mjs"
// @ts-check
import { defineConfig } from "astro/config";
import tailwindcss from "@tailwindcss/vite";
import press from "@fumapress/astro";

export default defineConfig({
  // the URL of your deployed site, for the sitemap and other full URLs
  // site: "https://example.com",
  integrations: [
    press({
      collections: { docs: { dir: "content" } },
      lastModified: true,
      includeProcessedMarkdown: true,
    }),
  ],
  vite: {
    plugins: [tailwindcss()],
  },
});
```

```tsx tab="press.config.tsx"
import { getCollection } from "astro:content";
import { toFumadocsSource } from "@fumadocs/astro/source";
import { createPress } from "@fumapress/astro/config";

export default createPress(
  toFumadocsSource(await getCollection("docs"), await getCollection("meta")),
).config({
  site: {
    name: "My Docs",
  },
});
```

```ts tab="src/content.config.ts"
import { defineCollection } from "astro:content";
import { metaLoader, pageLoader } from "@fumadocs/astro/loaders";

export const collections = {
  docs: defineCollection({ loader: pageLoader() }),
  meta: defineCollection({ loader: metaLoader() }),
};
```

```css tab="src/app.css"
@import "@fumapress/astro/css/default.css";
```

```json tab="tsconfig.json"
{
  "extends": "astro/tsconfigs/strict",
  "include": [".astro/types.d.ts", "**/*"],
  "exclude": ["dist"],
  "compilerOptions": {
    "jsx": "react-jsx",
    "jsxImportSource": "react",
    "paths": { "@/*": ["./src/*"] }
  }
}
```

</show>

The [Fumadocs CLI](./layouts.mdx#customizing-fumadocs-ui) uses the `paths` alias in the files it copies.

### Update package.json [step]

<show when={props.framework === "waku"}>

```json title="package.json"
{
  // [!code ++]
  "type": "module",
  "scripts": {
    // [!code ++:4]
    "dev": "fumapress dev",
    "build": "fumapress build",
    "start": "fumapress start",
    "types:check": "tsc --noEmit"
  }
}
```

</show>

<show when={props.framework === "astro"}>

```json title="package.json"
{
  // [!code ++]
  "type": "module",
  "scripts": {
    // [!code ++:4]
    "dev": "astro dev",
    "build": "astro build",
    "preview": "astro preview",
    "types:check": "astro sync && tsc --noEmit"
  }
}
```

</show>

### Write Your First Page [step]

```mdx title="content/index.mdx"
---
title: My Page
description: Hello World
---

## Overview

This is my first document.
```

### Start the Site [step]

```npm
npm run dev
```

Your docs start at `/`. To keep `/` for a landing page, [give your content a URL prefix](./content-sources/collections.mdx#multiple-collections).

</show>
